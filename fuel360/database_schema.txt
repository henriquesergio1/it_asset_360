
-- =================================================================================
-- CLEAN INSTALL SCRIPT - FUEL360 DATABASE SCHEMA (PRODUÇÃO v2.7)
-- =================================================================================

USE [master];
GO

IF NOT EXISTS (SELECT * FROM sys.databases WHERE name = 'Fuel360')
BEGIN
    CREATE DATABASE Fuel360;
END
GO

USE [Fuel360];
GO

-- 1. LIMPEZA TOTAL (PARA EVITAR ERRO DE OBJETO JÁ EXISTENTE)
IF OBJECT_ID('LogsSistema', 'U') IS NOT NULL DROP TABLE LogsSistema;
IF OBJECT_ID('ReembolsoDiario', 'U') IS NOT NULL DROP TABLE ReembolsoDiario;
IF OBJECT_ID('ReembolsoDetalhe', 'U') IS NOT NULL DROP TABLE ReembolsoDetalhe;
IF OBJECT_ID('ReembolsoHistorico', 'U') IS NOT NULL DROP TABLE ReembolsoHistorico;
IF OBJECT_ID('RotaPrevistaDiario', 'U') IS NOT NULL DROP TABLE RotaPrevistaDiario;
IF OBJECT_ID('RotaPrevistaDetalhe', 'U') IS NOT NULL DROP TABLE RotaPrevistaDetalhe;
IF OBJECT_ID('RotaPrevistaHistorico', 'U') IS NOT NULL DROP TABLE RotaPrevistaHistorico;
IF OBJECT_ID('Ausencias', 'U') IS NOT NULL DROP TABLE Ausencias;
IF OBJECT_ID('Usuarios', 'U') IS NOT NULL DROP TABLE Usuarios;
IF OBJECT_ID('SystemSettings', 'U') IS NOT NULL DROP TABLE SystemSettings;
IF OBJECT_ID('Colaboradores', 'U') IS NOT NULL DROP TABLE Colaboradores;
IF OBJECT_ID('Grupos', 'U') IS NOT NULL DROP TABLE Grupos;
GO

-- 2. CRIAÇÃO DAS TABELAS
CREATE TABLE Grupos (
    ID_Grupo INT IDENTITY(1,1) PRIMARY KEY,
    Nome NVARCHAR(100) UNIQUE NOT NULL,
    DataCriacao DATETIME DEFAULT GETDATE()
);

CREATE TABLE Colaboradores (
    ID_Colaborador INT IDENTITY(1,1) PRIMARY KEY,
    ID_Pulsus INT UNIQUE NOT NULL, 
    CodigoSetor INT NOT NULL,
    Nome NVARCHAR(200) NOT NULL,
    Grupo NVARCHAR(100) NOT NULL, 
    TipoVeiculo NVARCHAR(50) CHECK (TipoVeiculo IN ('Carro', 'Moto')),
    Ativo BIT DEFAULT 1,
    EnderecoBase NVARCHAR(MAX),
    LatitudeBase DECIMAL(12, 9),
    LongitudeBase DECIMAL(12, 9),
    EnderecoPendente BIT DEFAULT 0, -- Novo v2.7: Marca se o endereço precisa ser revisto por troca de nome
    DataCriacao DATETIME DEFAULT GETDATE()
);

CREATE TABLE Usuarios (
    ID_Usuario INT IDENTITY(1,1) PRIMARY KEY,
    Nome NVARCHAR(200) NOT NULL,
    Usuario NVARCHAR(100) UNIQUE NOT NULL,
    SenhaHash NVARCHAR(MAX) NOT NULL,
    Perfil NVARCHAR(50) DEFAULT 'Admin',
    Ativo BIT DEFAULT 1
);

CREATE TABLE SystemSettings (
    ID INT PRIMARY KEY CHECK (ID = 1),
    CompanyName NVARCHAR(200) DEFAULT 'Fuel360 Enterprise',
    LogoUrl NVARCHAR(MAX),
    FuelPrice DECIMAL(10, 4) DEFAULT 5.89,
    KmL_Car INT DEFAULT 10,
    KmL_Moto INT DEFAULT 35,
    Alert_MaxDailyKM INT DEFAULT 400,
    Alert_MaxClientDist INT DEFAULT 100,
    
    -- Integração MariaDB (Colaboradores)
    ExtDb_Host NVARCHAR(255),
    ExtDb_Port INT DEFAULT 3306,
    ExtDb_User NVARCHAR(100),
    ExtDb_Pass NVARCHAR(255),
    ExtDb_Database NVARCHAR(100),
    ExtDb_Query NVARCHAR(MAX),

    -- Integração SQL Server (Roteiro)
    ExtRoute_Host NVARCHAR(255),
    ExtRoute_Port INT DEFAULT 1433,
    ExtRoute_User NVARCHAR(100),
    ExtRoute_Pass NVARCHAR(255),
    ExtRoute_Database NVARCHAR(100),
    ExtRoute_Query NVARCHAR(MAX),

    -- Licenciamento
    LicenseKey NVARCHAR(MAX),
    LicenseClient NVARCHAR(200),
    LicenseExpires DATETIME
);

CREATE TABLE ReembolsoHistorico (
    ID_Historico INT IDENTITY(1,1) PRIMARY KEY,
    Periodo NVARCHAR(100) NOT NULL,
    DataFechamento DATETIME DEFAULT GETDATE(),
    TotalGeral DECIMAL(18, 2) NOT NULL,
    UsuarioFechamento NVARCHAR(100),
    OrigemDados NVARCHAR(50) DEFAULT 'CSV',
    MotivoEdicao NVARCHAR(MAX),
    ID_RotaHist INT
);

CREATE TABLE ReembolsoDetalhe (
    ID_Detalhe INT IDENTITY(1,1) PRIMARY KEY,
    ID_Historico INT NOT NULL,
    ID_Pulsus INT NOT NULL,
    NomeColaborador NVARCHAR(200),
    Grupo NVARCHAR(100),
    TipoVeiculo NVARCHAR(50),
    TotalKM DECIMAL(18, 4),
    ValorReembolso DECIMAL(18, 2),
    ParametroPreco DECIMAL(10, 4),
    ParametroKmL INT,
    Efetividade DECIMAL(5, 4),
    Ajuste DECIMAL(18, 2) DEFAULT 0,
    FOREIGN KEY (ID_Historico) REFERENCES ReembolsoHistorico(ID_Historico) ON DELETE CASCADE
);

CREATE TABLE ReembolsoDiario (
    ID_Diario INT IDENTITY(1,1) PRIMARY KEY,
    ID_Detalhe INT NOT NULL,
    DataOcorrencia DATE NOT NULL,
    KM_Dia DECIMAL(18, 4),
    Valor_Dia DECIMAL(18, 2),
    Observacao NVARCHAR(MAX),
    FOREIGN KEY (ID_Detalhe) REFERENCES ReembolsoDetalhe(ID_Detalhe) ON DELETE CASCADE
);

-- TABELAS PARA SIMULAÇÃO DO ROTEIRIZADOR
CREATE TABLE RotaPrevistaHistorico (
    ID_RotaHist INT IDENTITY(1,1) PRIMARY KEY,
    Periodo NVARCHAR(100),
    DataSimulacao DATETIME DEFAULT GETDATE(),
    TotalKM DECIMAL(18, 2),
    UsuarioSimulacao NVARCHAR(100)
);

CREATE TABLE RotaPrevistaDetalhe (
    ID_RotaDet INT IDENTITY(1,1) PRIMARY KEY,
    ID_RotaHist INT NOT NULL,
    ID_Pulsus INT,
    NomeColaborador NVARCHAR(200),
    Grupo NVARCHAR(100),
    TotalKM DECIMAL(18, 4),
    FOREIGN KEY (ID_RotaHist) REFERENCES RotaPrevistaHistorico(ID_RotaHist) ON DELETE CASCADE
);

CREATE TABLE RotaPrevistaDiario (
    ID_RotaDia INT IDENTITY(1,1) PRIMARY KEY,
    ID_RotaDet INT NOT NULL,
    DataVisita DATE,
    KM_Dia DECIMAL(18, 4),
    FOREIGN KEY (ID_RotaDet) REFERENCES RotaPrevistaDetalhe(ID_RotaDet) ON DELETE CASCADE
);

CREATE TABLE Ausencias (
    ID_Ausencia INT IDENTITY(1,1) PRIMARY KEY,
    ID_Colaborador INT NOT NULL,
    DataInicio DATE NOT NULL,
    DataFim DATE NOT NULL,
    Motivo NVARCHAR(200) NOT NULL,
    FOREIGN KEY (ID_Colaborador) REFERENCES Colaboradores(ID_Colaborador)
);

CREATE TABLE LogsSistema (
    ID_Log INT IDENTITY(1,1) PRIMARY KEY,
    DataHora DATETIME DEFAULT GETDATE(),
    Usuario NVARCHAR(100),
    Acao NVARCHAR(100),
    Detalhes NVARCHAR(MAX)
);
GO
